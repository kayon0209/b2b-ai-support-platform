# Runtime dependencies for the B2B AI support platform.
#
# This is what the API, the workers and the containers need to run. It exists
# as a separate file from `requirements-dev.txt` for one reason: the API image
# used to hand-list its own subset of packages, and drifted. It omitted
# `prometheus-client`, so `platform_core.main` raised ModuleNotFoundError on
# import and the container could not start at all - while the test suite, which
# runs against the venv, stayed green.
#
# One manifest, included by the other, so a package added here is in the image
# and in the dev environment at the same time.
#
# Ranges rather than exact pins: this is a pre-1.0 platform and the locks are
# maintained by the CI image, not here. Lower bounds are the versions the suite
# was actually verified against.

# --- Web framework and configuration ---------------------------------------
fastapi>=0.141.1
uvicorn[standard]>=0.32
pydantic>=2.9
pydantic-settings>=2.6
# Required by FastAPI to parse multipart/form-data, which the knowledge
# document upload uses. Without it every Form/File route raises at import
# time, so it is a runtime dependency, not an optional extra.
python-multipart>=0.0.9

# --- Persistence -----------------------------------------------------------
sqlalchemy>=2.0.36
psycopg[binary,pool]>=3.2
alembic>=1.14
greenlet>=3.1          # SQLAlchemy's async bridge; required explicitly on 3.12
pgvector>=0.3.6

# --- Async work and caching ------------------------------------------------
# The custom platform owns its own Redis. It is never shared with Chatwoot
# (see AGENTS.md, "Prohibited shortcuts").
redis>=5.2
celery>=5.4

# --- Object storage --------------------------------------------------------
minio>=7.2

# --- Security --------------------------------------------------------------
pyjwt[crypto]>=2.14.0
cryptography>=43
# SAML response signature verification. A vetted XML-DSig implementation
# rather than a hand-rolled one: getting canonicalisation or the
# signature-wrapping defence subtly wrong is how a SAML SP accepts a
# forged assertion, and that is not a bug a test suite reliably finds.
signxml>=4
lxml>=5

# --- Observability ---------------------------------------------------------
structlog>=24.4
# Imported by platform_core.http_metrics at module import, so the API cannot
# start without it. This is the package the Dockerfile was missing.
prometheus-client>=0.21
# Imported by `observability_tracing.build_tracer` as soon as
# `OTEL_EXPORTER_OTLP_ENDPOINT` is set, and the module handles the resulting
# `ImportError` by falling back to an in-process ring buffer. That fallback is
# why the omission was invisible: with neither package declared, setting the
# endpoint logged a warning and no span was ever exported, in every deployment.
#
# The floor is the first release whose `OTLPSpanExporter` and `Resource` import
# paths this module uses; the four imports were verified against 1.44.0.
opentelemetry-sdk>=1.27
opentelemetry-exporter-otlp>=1.27

# --- Contracts and utilities -----------------------------------------------
jsonschema>=4.23
pyyaml>=6.0
httpx>=0.27
tenacity>=9.0
uuid6>=2025.0.1

# --- Document parsing -------------------------------------------------------
# Used by the ingestion pipeline to extract text from PDF and DOCX uploads.
pypdf>=5.0
python-docx>=1.2
